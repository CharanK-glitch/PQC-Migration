\section{Mathematical Abbreviations Glossary \& Connecting the Dots}

\subsection{1. Variable Abbreviations Glossary}
\begin{itemize}
    \item $S_{\text{base}}$: \textbf{Base Score} (0--100). The raw safety ratio of quantum-resistant assets.
    \item $W_{\text{safe}}$: \textbf{Weighted Safe Assets}. Accumulated weight score of quantum-resistant algorithms.
    \item $W_{\text{total}}$: \textbf{Total Weighted Assets}. Sum of weights across ALL detected cryptographic calls.
    \item $w_{\text{ctx}}(i)$: \textbf{Context Weight}. Threat level of primitive $i$ (Key-Exchange/Signatures = $1.0$, Encryption = $0.7$, Hashes = $0.4$).
    \item $c_i$: \textbf{Occurrence Count}. Total frequency of algorithm $i$ calls found in source code.
    \item $B_{\text{PQC}}$: \textbf{PQC Adoption Bonus} ($+0$ to $+5$ pts). Reward for utilizing NIST FIPS 203/204 algorithms.
    \item $P_{\text{critical}}$: \textbf{Critical Vulnerability Penalty} ($-0$ to $-15$ pts). Deduction for Category 0 algorithms (RSA, ECDSA).
    \item $P_{\text{diversity}}$: \textbf{Polyglot Complexity Penalty} ($-0$ to $-5$ pts). Deduction for codebases spanning $> 2$ programming languages.
    \item $P_{\text{PG}}$: \textbf{PostgreSQL Penalty} ($-0$ to $-10$ pts). Deduction for database security flaws (\texttt{pgcrypto} RSA, MD5).
\end{itemize}

\subsection{2. Connecting the Dots: Step-by-Step Mathematical Flow}
\begin{enumerate}
    \item \textbf{Asset Weighting}: Each detected code call accumulates impact score $= w_{\text{ctx}} \times c_i$. Safe calls contribute to $W_{\text{safe}}$; all calls to $W_{\text{total}}$.
    \item \textbf{Base Safety Ratio}: Computes baseline percentage $S_{\text{base}} = \frac{W_{\text{safe}}}{W_{\text{total}}} \times 100$.
    \item \textbf{Applying Bonus}: If $N_{\text{pqc}} > 0$, adds $B_{\text{PQC}} = \min\left(5, \frac{N_{\text{pqc}}}{N_{\text{assets}}} \times 10\right)$.
    \item \textbf{Applying Penalties}: Deducts $P_{\text{critical}}$ for classical keys, $P_{\text{diversity}}$ for multi-language complexity, and $P_{\text{PG}}$ for database risks.
    \item \textbf{Final Clamping}: Final Score $= \max\left(0, \min\left(100, \left\lfloor S_{\text{base}} + B_{\text{PQC}} - P_{\text{critical}} - P_{\text{diversity}} - P_{\text{PG}} \right\rceil\right)\right)$.
\end{enumerate}

\subsection{3. Pre-existing Codebase Formulas \& Equations}
\[
\text{Final Score} = \max\left(0, \min\left(100, \left\lfloor S_{\text{base}} + B_{\text{PQC}} - P_{\text{critical}} - P_{\text{diversity}} - P_{\text{PG}} \right\rceil\right)\right)
\]
\[
S_{\text{base}} = \frac{W_{\text{safe}}}{W_{\text{total}}} \times 100 = \frac{\sum_{i \in \text{Safe}} w_{\text{ctx}}(i) \cdot c_i}{\sum_{j \in \text{All}} w_{\text{ctx}}(j) \cdot c_j} \times 100
\]
\begin{align*}
B_{\text{PQC}} &= \min\left(5, \; \frac{N_{\text{pqc}}}{N_{\text{assets}}} \times 10\right) \qquad P_{\text{critical}} = \min\left(15, \; \frac{N_{\text{critical}}}{N_{\text{assets}}} \times 20\right) \\
P_{\text{diversity}} &= \min\left(5, \; \max\left(0, (L - 2) \times 1.5\right)\right) \qquad P_{\text{PG}} = \min\left(10, \; \sum \text{RulePenalties}\right)
\end{align*}

The exact numerical coefficients (e.g., $\min(5, \dots)$, $1.5$ multiplier) do not exist on a single webpage. Instead, they represent a custom algorithmic synthesis model that translates qualitative standards from NIST SP 800-57, NIST IR 8413, CISA Quantum Guidance, and PostgreSQL security docs into quantifiable linear equations.

\section{API Call Architecture \& Inter-Module Data Flow Between Python (.py) Files}

The \texttt{pgcrypto} scanner architecture connects Python modules via explicit API calls and internal data contracts:

\begin{enumerate}
    \item \textbf{CLI to Ingestion API} (\texttt{pgcrypto\_scanner.py} $\rightarrow$ \texttt{scanner/git\_cloner.py}):
    \begin{itemize}
        \item \textit{Function Call}: \texttt{RepoCloner.clone\_repo(github\_url, timeout=300)}
        \item \textit{Action}: Invokes Git/GitHub HTTPS API via sub-process to execute \texttt{git clone --depth 1}. Returns local path to cloned workspace.
    \end{itemize}
    \item \textbf{Ingestion to Detection API} (\texttt{pgcrypto\_scanner.py} $\rightarrow$ \texttt{scanner/repo\_detector.py}):
    \begin{itemize}
        \item \textit{Function Call}: \texttt{RepoDetector.detect(workspace\_dir)}
        \item \textit{Action}: Inspects directory tree to auto-detect PostgreSQL configs, SQL migration scripts, or C++/Rust/Python codebases.
    \end{itemize}
    \item \textbf{Scanner Dispatch APIs} (\texttt{pgcrypto\_scanner.py} $\rightarrow$ \texttt{scanner/postgres\_scanner.py} \& \texttt{generic\_scanner.py}):
    \begin{itemize}
        \item \textit{Function Call}: \texttt{PostgresScanner.scan\_directory()} \& \texttt{GenericScanner.scan\_directory()}
        \item \textit{Internal File I/O API}: Uses Python \texttt{open(filepath, "r", encoding="utf-8", errors="ignore")} line-by-line.
        \item \textit{Rule Evaluation}: Matches line content against Regex rules imported from \texttt{rules/postgres/*.py} and \texttt{rules/generic/*.py}.
        \item \textit{Return Contract}: Returns raw list of finding dicts containing \texttt{algorithmName}, \texttt{primitive}, \texttt{file}, \texttt{line\_number}, and \texttt{risk}.
    \end{itemize}
    \item \textbf{Deduplication \& Normalization API} (\texttt{pgcrypto\_scanner.py} $\rightarrow$ \texttt{scanner/merger.py}):
    \begin{itemize}
        \item \textit{Function Call}: \texttt{merge\_findings(pg\_findings, generic\_findings)} \& \texttt{deduplicate\_by\_algorithm()}
        \item \textit{Action}: Combines findings and groups occurrences per unique algorithm family to prevent duplicate counting.
    \end{itemize}
    \item \textbf{Risk \& Scoring Evaluation APIs} (\texttt{pgcrypto\_scanner.py} $\rightarrow$ \texttt{scoring/weight_engine.py} \& \texttt{nist_compliance.py}):
    \begin{itemize}
        \item \textit{Function Call}: \texttt{compute\_readiness_score(findings)} \& \texttt{check\_nist\_compliance(findings)}
        \item \textit{Action}: Evaluates Base Score, Bonuses, Penalties, and NIST FIPS 203/204 compliance. Returns structured score dictionary.
    \end{itemize}
    \item \textbf{Output Generator APIs} (\texttt{pgcrypto\_scanner.py} $\rightarrow$ \texttt{output/cbom_generator.py} \& \texttt{output/score_reporter.py}):
    \begin{itemize}
        \item \textit{Function Call}: \texttt{generate\_cbom(findings)} $\rightarrow$ \texttt{save\_cbom("cbom.json")}
        \item \textit{Function Call}: \texttt{generate\_score\_report(score_dict)} $\rightarrow$ \texttt{save\_score\_report("readiness\_score.json")} \& \texttt{print\_terminal\_report()}
        \item \textit{Dashboard Call}: Renders interactive HTML web dashboard into \texttt{dashboard/index.html}.
    \end{itemize}
\end{enumerate}
